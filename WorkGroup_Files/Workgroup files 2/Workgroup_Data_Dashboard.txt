const ScoreBadge = ({ score }) => {
  let bgColor = "bg-gray-100 text-gray-800";
  if (score >= 90) bgColor = "bg-green-100 text-green-800 border-green-200";
  else if (score >= 80) bgColor = "bg-blue-100 text-blue-800 border-blue-200";
  else if (score >= 70) bgColor = "bg-yellow-100 text-yellow-800 border-yellow-200";
  else bgColor = "bg-red-100 text-red-800 border-red-200";

  return (
    <span className={`px-3 py-1 rounded-full text-sm font-semibold border ${bgColor}`}>
      {score.toFixed(2)}
    </span>
  );
};
